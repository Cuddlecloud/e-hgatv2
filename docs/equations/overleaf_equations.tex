=
\frac{1}{n}
\sum_{i=1}^{n}

    \min_{j\in PF^{*}}\left\{d_{(i,j)}^{+}\right\}

\tag{22a}
\end{equation}

\begin{equation}

=
\sqrt{
    \left(\max\left\{f_1^{i}-f_1^{j},0\right\}\right)^2
    +

\tag{22b}
\end{equation}

\begin{equation}
\Delta(PF)
=
\frac{
    d_u+d_b+\sum_{i=1}^{n-1}\left|d_i-\bar{d}\right|

    d_u+d_b+\bar{d}(n-1)
}.
\tag{23}
\end{equation}

\begin{equation}

\frac{f_1-f_1^{\min}}{f_1^{\max}-f_1^{\min}},
